// Models staged for later phases. Not loaded by Prisma (.txt). Move each model into schema.prisma
// in the phase that needs it, and restore its back-relation on Workspace.

model Invite {
  id          String     @id @default(cuid())
  workspaceId String
  email       String
  role        MemberRole @default(VIEWER)
  token       String     @unique
  expiresAt   DateTime
  acceptedAt  DateTime?
  workspace   Workspace  @relation(fields: [workspaceId], references: [id], onDelete: Cascade)

  @@index([workspaceId])
}

model StripeEvent {
  id          String   @id // Stripe event id, for webhook idempotency
  type        String
  processedAt DateTime @default(now())
}

enum UsageKind {
  COVER_SCAN
  IMPORT
}

model UsageEvent {
  id          String    @id @default(cuid())
  workspaceId String
  kind        UsageKind
  createdAt   DateTime  @default(now())
  workspace   Workspace @relation(fields: [workspaceId], references: [id], onDelete: Cascade)

  @@index([workspaceId, kind, createdAt])
}

// ---------- Lending ----------

model Borrower {
  id          String    @id @default(cuid())
  workspaceId String
  name        String
  phone       String?
  email       String?
  memberCode  String?
  notes       String?
  createdAt   DateTime  @default(now())
  workspace   Workspace @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  loans       Loan[]

  @@index([workspaceId])
  @@unique([workspaceId, memberCode])
}

model Loan {
  id          String    @id @default(cuid())
  workspaceId String
  copyId      String
  borrowerId  String
  lentAt      DateTime  @default(now())
  dueAt       DateTime?
  returnedAt  DateTime?
  notes       String?
  copy        Copy      @relation(fields: [copyId], references: [id], onDelete: Cascade)
  borrower    Borrower  @relation(fields: [borrowerId], references: [id], onDelete: Cascade)

  @@index([workspaceId, returnedAt])
  @@index([copyId])

}
