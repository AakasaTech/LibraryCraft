// Models staged for later phases. Not loaded by Prisma (.txt). Move each model into schema.prisma
// in the phase that needs it, and restore its back-relation on Workspace.

model Invite {
  id          String     @id @default(cuid())
  workspaceId String
  email       String
  role        MemberRole @default(VIEWER)
  token       String     @unique
  expiresAt   DateTime
  acceptedAt  DateTime?
  workspace   Workspace  @relation(fields: [workspaceId], references: [id], onDelete: Cascade)

  @@index([workspaceId])
}

model StripeEvent {
  id          String   @id // Stripe event id, for webhook idempotency
  type        String
  processedAt DateTime @default(now())
}

enum UsageKind {
  COVER_SCAN
  IMPORT
}

model UsageEvent {
  id          String    @id @default(cuid())
  workspaceId String
  kind        UsageKind
  createdAt   DateTime  @default(now())
  workspace   Workspace @relation(fields: [workspaceId], references: [id], onDelete: Cascade)

  @@index([workspaceId, kind, createdAt])
}

// ---------- Catalogue ----------

enum EditionSource {
  OPENLIBRARY
  GOOGLE
  MANUAL
}

model Edition {
  id        String        @id @default(cuid())
  isbn10    String?       @unique
  isbn13    String?       @unique
  title     String
  subtitle  String?
  authors   String[]
  publisher String?
  year      Int?
  language  String?
  pages     Int?
  coverUrl  String?
  source    EditionSource
  createdAt DateTime      @default(now())
  copies    Copy[]

  @@index([title])
}

enum ReadStatus {
  UNREAD
  READING
  READ
}

model Room {
  id          String     @id @default(cuid())
  workspaceId String
  name        String
  position    Int        @default(0)
  workspace   Workspace  @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  bookcases   Bookcase[]

  @@index([workspaceId])
}

model Bookcase {
  id          String  @id @default(cuid())
  workspaceId String
  roomId      String
  name        String
  position    Int     @default(0)
  room        Room    @relation(fields: [roomId], references: [id], onDelete: Cascade)
  shelves     Shelf[]

  @@index([workspaceId])
}

model Shelf {
  id          String   @id @default(cuid())
  workspaceId String
  bookcaseId  String
  name        String
  position    Int      @default(0)
  bookcase    Bookcase @relation(fields: [bookcaseId], references: [id], onDelete: Cascade)
  copies      Copy[]

  @@index([workspaceId])
}

model Copy {
  id          String     @id @default(cuid())
  workspaceId String
  editionId   String
  shelfId     String?
  condition   String?
  readStatus  ReadStatus @default(UNREAD)
  notes       String?
  addedAt     DateTime   @default(now())
  workspace   Workspace  @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  edition     Edition    @relation(fields: [editionId], references: [id])
  shelf       Shelf?     @relation(fields: [shelfId], references: [id], onDelete: SetNull)
  tags        CopyTag[]
  loans       Loan[]

  @@index([workspaceId])
  @@index([workspaceId, editionId])
  @@index([shelfId])
}

model Tag {
  id          String    @id @default(cuid())
  workspaceId String
  name        String
  workspace   Workspace @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  copies      CopyTag[]

  @@unique([workspaceId, name])
}

model CopyTag {
  copyId String
  tagId  String
  copy   Copy   @relation(fields: [copyId], references: [id], onDelete: Cascade)
  tag    Tag    @relation(fields: [tagId], references: [id], onDelete: Cascade)

  @@id([copyId, tagId])
}

// ---------- Lending ----------

model Borrower {
  id          String    @id @default(cuid())
  workspaceId String
  name        String
  phone       String?
  email       String?
  memberCode  String?
  notes       String?
  createdAt   DateTime  @default(now())
  workspace   Workspace @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  loans       Loan[]

  @@index([workspaceId])
  @@unique([workspaceId, memberCode])
}

model Loan {
  id          String    @id @default(cuid())
  workspaceId String
  copyId      String
  borrowerId  String
  lentAt      DateTime  @default(now())
  dueAt       DateTime?
  returnedAt  DateTime?
  notes       String?
  copy        Copy      @relation(fields: [copyId], references: [id], onDelete: Cascade)
  borrower    Borrower  @relation(fields: [borrowerId], references: [id], onDelete: Cascade)

  @@index([workspaceId, returnedAt])
  @@index([copyId])

}
